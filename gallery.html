const form = document.querySelector("#reservationForm"); const successMessage = document.querySelector("#successMessage"); const dateInput = document.querySelector("#date"); /* Sætter minimumsdato til i dag */ const today = new Date().toISOString().split("T")[0]; dateInput.setAttribute("min", today); form.addEventListener("submit", function (event) { event.preventDefault(); /* Tjekker om formularen er valid */ if (!form.checkValidity()) { form.classList.add("was-validated"); successMessage.classList.add("d-none"); return; } /* Viser success-besked */ form.classList.remove("was-validated"); successMessage.classList.remove("d-none"); /* Nulstiller formular */ form.reset(); /* Scroller til beskeden */ successMessage.scrollIntoView({ behavior: "smooth", block: "center", }); });
